    77→/* ── Animated number (framer-motion value rendered as text) ──────────────── */
    78→
    79→function CountUp({ value, format }: { value: number; format: (n: number) => string }) {
    80→  const mv = useMotionValue(0)
    81→  const text = useTransform(mv, format)
    82→  useEffect(() => {
    83→    const controls = animate(mv, value, { duration: 0.8, ease: 'easeOut' })
    84→    return () => controls.stop()
    85→  }, [mv, value])
    86→  return <motion.span>{text}</motion.span>
    87→}

   733→/* ── P&L strip: revenue − expenses = net profit ─────────────────────────── */
   734→
   735→function ProfitStrip({
   736→  revenue,
   737→  expenses,
   738→  onManageExpenses,
   739→}: {
   740→  revenue: number
   741→  expenses: { total: number; count: number; byCategory: Record<string, { amount: number; count: number }> }
   742→  onManageExpenses: () => void
   743→}) {
   744→  const profit = round2(revenue - expenses.total)
   745→  const margin = revenue > 0 ? Math.round((profit / revenue) * 100) : 0
   746→  const cats = Object.entries(expenses.byCategory).sort((a, b) => b[1].amount - a[1].amount).slice(0, 4)
   747→  return (
   748→    <motion.div
   749→      initial={{ opacity: 0, y: 10 }}
   750→      animate={{ opacity: 1, y: 0 }}
   751→      transition={{ delay: 0.1, duration: 0.3 }}
   752→      className="hairline mt-4 rounded-xl bg-card p-4 lg:p-5"
   753→    >
   754→      <div className="flex flex-wrap items-center justify-between gap-3">
   755→        <h2 className="flex items-center gap-2 text-sm font-semibold">
   756→          <TrendingUp className="size-4 text-primary" aria-hidden />
   757→          Profit & Loss
   758→        </h2>
   759→        <Button variant="outline" size="sm" className="h-9 gap-1.5" onClick={onManageExpenses}>
   760→          <ReceiptText className="size-3.5" aria-hidden />
   761→          Manage Expenses
   762→        </Button>
   763→      </div>
   764→      <div className="mt-4 grid grid-cols-3 items-center gap-2 sm:gap-4">
   765→        <div>
   766→          <p className="micro text-muted-foreground">Revenue</p>
   767→          <p className="tnum mt-0.5 text-base font-bold sm:text-xl">{formatINR(revenue)}</p>
   768→        </div>
   769→        <div className="text-center">
   770→          <p className="micro text-muted-foreground">Expenses</p>
   771→          <p className="tnum mt-0.5 text-base font-bold text-rose-600 sm:text-xl dark:text-rose-400">
   772→            −{formatINR(expenses.total)}
   773→          </p>
   774→        </div>
   775→        <div className="text-right">
   776→          <p className="micro text-muted-foreground">Net Profit</p>
   777→          <p
   778→            className={`tnum mt-0.5 text-base font-bold sm:text-xl ${
   779→              profit >= 0 ? 'text-emerald-600 dark:text-emerald-400' : 'text-destructive'
   780→            }`}
   781→          >
   782→            {formatINR(profit)}
   783→          </p>
   784→        </div>
   785→      </div>
   786→      {/* Expense share of revenue bar */}
   787→      <div className="mt-3 flex h-2.5 overflow-hidden rounded-full bg-muted" aria-hidden>
   788→        {expenses.total > 0 && revenue > 0 && (
   789→          <div
   790→            className="h-full bg-rose-500/70"
   791→            style={{ width: `${Math.min(96, Math.round((expenses.total / revenue) * 100))}%` }}
   792→          />
   793→        )}
   794→      </div>
   795→      <div className="mt-3 flex flex-wrap items-center gap-x-4 gap-y-1.5 text-xs text-muted-foreground">
   796→        <span>
   797→          Margin:{' '}
   798→          <span className={`tnum font-bold ${profit >= 0 ? 'text-emerald-600 dark:text-emerald-400' : 'text-destructive'}`}>
   799→            {margin}%
   800→          </span>
   801→        </span>
   802→        {cats.map(([cat, v]) => (
   803→          <span key={cat} className="tnum flex items-center gap-1">
   804→            <span className="size-1.5 rounded-full bg-rose-500/60" aria-hidden />
   805→            {cat.charAt(0) + cat.slice(1).toLowerCase()} {formatINR(v.amount)}
   806→          </span>
   807→        ))}
   808→        {expenses.count === 0 && (
   809→          <span className="italic">
   810→            No expenses logged yet — add rent, supplies & salaries for a true profit line.
   811→          </span>
   812→        )}
   813→      </div>
   814→    </motion.div>
   815→  )
   816→}
