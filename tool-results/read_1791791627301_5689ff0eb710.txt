   351→export default function TransactionsView() {
   352→  const user = useAppStore((s) => s.user)
   353→  const [range, setRange] = useState<DateRange>(() => computePreset('30d'))
   354→  const [method, setMethod] = useState('')
   355→  const [staffId, setStaffId] = useState('all')
   356→  const [search, setSearch] = useState('')
   357→  const [selected, setSelected] = useState<TransactionDTO | null>(null)
   358→  const debouncedSearch = useDebouncedValue(search, 400)
   359→
   360→  const staffQ = useQuery({
   361→    queryKey: ['staff'],
   362→    queryFn: () => apiGet<StaffListResponse>('/api/staff'),
   363→  })
   364→  const settingsQ = useQuery({
   365→    queryKey: ['settings'],
   366→    queryFn: () => apiGet<SettingsDTO>('/api/settings'),
   367→  })
   368→
   369→  const listQ = useInfiniteQuery({
   370→    queryKey: ['transactions-list', range.from, range.to, method, staffId, debouncedSearch],
   371→    queryFn: ({ pageParam }) => {
   372→      const p = new URLSearchParams({
   373→        from: range.from,
   374→        to: range.to,
   375→        page: String(pageParam),
   376→        limit: '25',
   377→      })
   378→      if (method) p.set('method', method)
   379→      if (staffId !== 'all') p.set('staffId', staffId)
   380→      if (debouncedSearch) p.set('search', debouncedSearch)
   381→      return apiGet<TransactionsResponse>(`/api/transactions?${p.toString()}`)
   382→    },
   383→    initialPageParam: 1,
   384→    getNextPageParam: (lastPage, allPages) => {
   385→      const loaded = allPages.reduce((n, pg) => n + pg.data.length, 0)
   386→      return loaded < lastPage.total ? allPages.length + 1 : undefined
   387→    },
   388→  })
   389→
   390→  const rows = useMemo(
   391→    () => (listQ.data ? listQ.data.pages.flatMap((p) => p.data) : []),
   392→    [listQ.data],
   393→  )
   394→  const summary = listQ.data?.pages[0]?.summary
   395→  const total = listQ.data?.pages[0]?.total ?? 0
   396→
   397→  function resetFilters() {
   398→    setMethod('')
   399→    setStaffId('all')
   400→    setSearch('')
   401→    setRange(computePreset('30d'))
   402→  }
   403→
   404→  function exportCsv() {
   405→    toast('Preparing CSV…')
   406→    window.open(
   407→      `/api/export/csv?from=${range.from}&to=${range.to}${method ? `&method=${method}` : ''}`,
   408→      '_blank',
   409→    )
   410→  }
   411→
   412→  function exportPdf() {
   413→    if (!rows.length || !summary || !settingsQ.data) return
   414→    // PDF mirrors the revenue summary: voided bills are excluded (they stay in
   415→    // the ledger + CSV export for the full audit trail).
   416→    const paidRows = rows.filter((r) => r.status !== 'VOIDED')
   417→    if (!paidRows.length) {
   418→      toast.error('All loaded bills are voided — nothing to export')
   419→      return
   420→    }
   421→    toast('Preparing PDF…')
   422→    try {
   423→      downloadReportPdf(paidRows, summary, settingsQ.data, range.from, range.to)
   424→    } catch (e) {
   425→      toast.error((e as Error).message)
   426→    }
   427→  }
   428→
   429→  const pdfDisabled = rows.length === 0 || !settingsQ.data
   430→
   431→  return (
   432→    <div className="p-4 pb-24 lg:p-6 lg:pb-6">
   433→      {/* Sticky filter bar */}
   434→      <div className="sticky top-16 z-20 -mx-4 -mt-4 bg-background/95 px-4 pt-4 pb-3 backdrop-blur lg:-mx-6 lg:-mt-6 lg:px-6 lg:pt-6 lg:pb-4">
   435→        <div className="flex flex-wrap items-center justify-between gap-2">
   436→          <div>
   437→            <h1 className="text-xl font-bold tracking-tight lg:text-2xl">Transactions</h1>
   438→            <p className="mt-0.5 text-sm text-muted-foreground">Search, filter and export bills</p>
   439→          </div>
   440→          <div className="flex items-center gap-2">
   441→            <Tooltip>
   442→              <TooltipTrigger asChild>
   443→                <Button variant="outline" size="sm" onClick={exportCsv}>
   444→                  <Download aria-hidden />
   445→                  <span className="hidden sm:inline">CSV</span>
   446→                </Button>
   447→              </TooltipTrigger>
   448→              <TooltipContent>Export filtered bills as CSV</TooltipContent>
   449→            </Tooltip>
   450→            <Tooltip>
   451→              <TooltipTrigger asChild>
   452→                <span className="inline-flex">
   453→                  <Button variant="outline" size="sm" onClick={exportPdf} disabled={pdfDisabled}>
   454→                    <FileDown aria-hidden />
   455→                    <span className="hidden sm:inline">PDF</span>
   456→                  </Button>
   457→                </span>
   458→              </TooltipTrigger>
   459→              <TooltipContent>
   460→                {pdfDisabled ? 'No bills to export' : 'Download PDF report of loaded bills'}
   461→              </TooltipContent>
   462→            </Tooltip>
   463→          </div>
   464→        </div>
   465→
   466→        <div className="mt-3 rounded-2xl border bg-card p-3 shadow-sm">
   467→          <div className="flex flex-wrap items-center gap-2">
   468→            <RangePicker value={range} onChange={setRange} />
   469→            <ToggleGroup
   470→              type="single"
   471→              variant="outline"
   472→              size="sm"
   473→              className="rounded-lg"
   474→              value={method || 'ALL'}
   475→              onValueChange={(v) => setMethod(v === 'ALL' || v === '' ? '' : v)}
   476→            >
   477→              <ToggleGroupItem value="ALL" className="px-2.5 text-xs">
   478→                All
   479→              </ToggleGroupItem>
   480→              <ToggleGroupItem value="CASH" className="gap-1 px-2.5 text-xs">
   481→                <Banknote className="size-3.5" aria-hidden />
   482→                Cash
   483→              </ToggleGroupItem>
   484→              <ToggleGroupItem value="UPI" className="gap-1 px-2.5 text-xs">
   485→                <QrCode className="size-3.5" aria-hidden />
   486→                UPI
   487→              </ToggleGroupItem>
   488→              <ToggleGroupItem value="CARD" className="gap-1 px-2.5 text-xs">
   489→                <CreditCard className="size-3.5" aria-hidden />
   490→                Card
   491→              </ToggleGroupItem>
   492→            </ToggleGroup>
   493→            <Select value={staffId} onValueChange={setStaffId}>
   494→              <SelectTrigger size="sm" className="w-[9.5rem]" aria-label="Filter by staff">
   495→                <SelectValue placeholder="All staff" />
   496→              </SelectTrigger>
   497→              <SelectContent>
   498→                <SelectItem value="all">All staff</SelectItem>
   499→                {staffQ.data?.users.map((u) => (
   500→                  <SelectItem key={u.id} value={u.id}>
   501→                    {u.name}
   502→                  </SelectItem>
   503→                ))}
   504→              </SelectContent>
   505→            </Select>
   506→            <div className="relative min-w-[150px] flex-1 sm:max-w-xs">
   507→              <Search
   508→                className="absolute top-1/2 left-2.5 size-4 -translate-y-1/2 text-muted-foreground"
   509→                aria-hidden
   510→              />
   511→              <Input
   512→                value={search}
   513→                onChange={(e) => setSearch(e.target.value)}
   514→                placeholder="Bill no, mobile or name"
   515→                className="h-8 pl-8 text-sm"
   516→                aria-label="Search bills"
   517→              />
   518→            </div>
   519→            <Button variant="ghost" size="sm" onClick={resetFilters}>
   520→              <RotateCcw aria-hidden />
   521→              Reset
   522→            </Button>
   523→          </div>
   524→        </div>
   525→      </div>
   526→
   527→      {/* Summary strip */}
   528→      {listQ.isPending ? (
   529→        <Skeleton className="mt-4 h-[5.5rem] rounded-2xl lg:mt-5" />
   530→      ) : summary ? (
   531→        <div className="mt-4 grid grid-cols-2 gap-3 rounded-2xl bg-muted/60 p-3 sm:grid-cols-4 lg:mt-5 lg:p-4">
   532→          <div>
   533→            <p className="text-[11px] font-medium tracking-wide text-muted-foreground uppercase">Revenue</p>
   534→            <p className="mt-0.5 text-lg font-bold tabular-nums">{formatINR(summary.revenue)}</p>
   535→          </div>
   536→          <div>
   537→            <p className="text-[11px] font-medium tracking-wide text-muted-foreground uppercase">Bills</p>
   538→            <p className="mt-0.5 text-lg font-bold tabular-nums">{summary.count.toLocaleString('en-IN')}</p>
   539→          </div>
   540→          <div>
   541→            <p className="text-[11px] font-medium tracking-wide text-muted-foreground uppercase">Avg Bill</p>
   542→            <p className="mt-0.5 text-lg font-bold tabular-nums">{formatINR(summary.avgBill)}</p>
   543→          </div>
   544→          <div>
   545→            <p className="text-[11px] font-medium tracking-wide text-muted-foreground uppercase">Payment Split</p>
   546→            <div className="mt-1 space-y-0.5">
   547→              {(['CASH', 'UPI', 'CARD'] as PaymentMethod[]).map((m) => (
   548→                <div key={m} className="flex items-center gap-1.5 text-[11px] text-muted-foreground">
   549→                  <span className={cn('size-1.5 shrink-0 rounded-full', METHOD_META[m].dotClass)} aria-hidden />
   550→                  <span className="w-7 shrink-0">{METHOD_META[m].label}</span>
   551→                  <span className="truncate font-medium text-foreground tabular-nums">
   552→                    {formatINR(summary.byMethod[m].amount)}
   553→                  </span>
   554→                  <span className="tabular-nums">({summary.byMethod[m].count})</span>
   555→                </div>
   556→              ))}
   557→            </div>
   558→          </div>
   559→        </div>
   560→      ) : null}
   561→
   562→      {/* Ledger */}
   563→      {listQ.isPending ? (
   564→        <div className="mt-4 space-y-2 lg:mt-5">
   565→          {Array.from({ length: 6 }).map((_, i) => (
   566→            <Skeleton key={i} className="h-16 rounded-2xl md:h-12" />
   567→          ))}
   568→        </div>
   569→      ) : listQ.isError ? (
   570→        <div className="mt-4 rounded-2xl border bg-card lg:mt-5">
   571→          <EmptyState icon={AlertTriangle} title="Could not load bills" hint={(listQ.error as Error).message} />
   572→          <div className="flex justify-center pb-6">
   573→            <Button variant="outline" onClick={() => void listQ.refetch()}>
   574→              Try again
   575→            </Button>
   576→          </div>
   577→        </div>
   578→      ) : rows.length === 0 ? (
   579→        <div className="mt-4 rounded-2xl border bg-card lg:mt-5">
   580→          <EmptyState
   581→            icon={ReceiptText}
   582→            title="No bills found"
   583→            hint="Try widening the date range or clearing the filters."
   584→          />
   585→        </div>
   586→      ) : (
   587→        <>
   588→          {/* Desktop table */}
   589→          <div className="mt-4 hidden overflow-hidden rounded-2xl border bg-card shadow-sm md:block lg:mt-5">
   590→            <Table>
   591→              <TableHeader>
   592→                <TableRow className="hover:bg-transparent">
   593→                  <TableHead className="pl-4">Bill No</TableHead>
   594→                  <TableHead>Date &amp; Time</TableHead>
   595→                  <TableHead>Customer</TableHead>
   596→                  <TableHead>Services</TableHead>
   597→                  <TableHead>Staff</TableHead>
   598→                  <TableHead>Method</TableHead>
   599→                  <TableHead className="pr-4 text-right">Total</TableHead>
   600→                </TableRow>
   601→              </TableHeader>
   602→              <TableBody>
   603→                {rows.map((t) => (
   604→                  <TableRow
   605→                    key={t.id}
   606→                    className={cn('cursor-pointer', t.status === 'VOIDED' && 'opacity-60')}
   607→                    onClick={() => setSelected(t)}
   608→                  >
   609→                    <TableCell className="pl-4 font-mono text-[13px] font-semibold">
   610→                      {t.billNo}
   611→                    </TableCell>
   612→                    <TableCell>
   613→                      <span className="block text-[13px]">{formatDate(t.createdAt)}</span>
   614→                      <span className="block text-[11px] text-muted-foreground">{formatTime(t.createdAt)}</span>
   615→                    </TableCell>
   616→                    <TableCell className="max-w-[160px]">
   617→                      <span className="block truncate text-[13px]">
   618→                        {t.customerName || t.customerMobile || (
   619→                          <span className="italic text-muted-foreground">Walk-in</span>
   620→                        )}
   621→                      </span>
   622→                    </TableCell>
   623→                    <TableCell className="max-w-[180px]">
   624→                      <span className="block truncate text-[13px]">{t.items[0]?.serviceName ?? '—'}</span>
   625→                      {t.items.length > 1 && (
   626→                        <span className="block text-[11px] text-muted-foreground">
   627→                          +{t.items.length - 1} more
   628→                        </span>
   629→                      )}
   630→                    </TableCell>
   631→                    <TableCell className="text-[13px]">{t.staffName}</TableCell>
   632→                    <TableCell>
   633→                      <div className="flex items-center gap-1.5">
   634→                        <MethodBadge method={t.paymentMethod} />
   635→                        {t.status === 'VOIDED' && <StatusBadge status={t.status} />}
   636→                      </div>
   637→                    </TableCell>
   638→                    <TableCell
   639→                      className={cn(
   640→                        'pr-4 text-right font-semibold tabular-nums',
   641→                        t.status === 'VOIDED' && 'text-muted-foreground line-through',
   642→                      )}
   643→                    >
   644→                      {formatINR(t.total)}
   645→                    </TableCell>
   646→                  </TableRow>
   647→                ))}
   648→              </TableBody>
   649→            </Table>
   650→          </div>
   651→
   652→          {/* Mobile cards */}
   653→          <div className="mt-4 space-y-2.5 md:hidden">
   654→            {rows.map((t, i) => (
   655→              <motion.button
   656→                key={t.id}
   657→                type="button"
   658→                initial={{ opacity: 0, y: 6 }}
   659→                animate={{ opacity: 1, y: 0 }}
   660→                transition={{ delay: Math.min(i, 6) * 0.03, duration: 0.22 }}
   661→                onClick={() => setSelected(t)}
   662→                className="w-full rounded-2xl border bg-card p-3.5 text-left shadow-sm transition-colors hover:bg-accent/50 active:bg-accent"
   663→              >
   664→                <div className="flex items-center justify-between gap-2">
   665→                  <p className="font-mono text-sm font-semibold">{t.billNo}</p>
   666→                  <p
   667→                    className={cn(
   668→                      'text-base font-bold tabular-nums',
   669→                      t.status === 'VOIDED' && 'text-muted-foreground line-through',
   670→                    )}
   671→                  >
   672→                    {formatINR(t.total)}
   673→                  </p>
   674→                </div>
   675→                <div className="mt-2 flex flex-wrap items-center gap-x-3 gap-y-1.5 text-xs text-muted-foreground">
   676→                  <span>
   677→                    {formatDate(t.createdAt)}, {formatTime(t.createdAt)}
   678→                  </span>
   679→                  <span className="max-w-[45%] truncate">
   680→                    {t.customerName || t.customerMobile || <span className="italic">Walk-in</span>}
   681→                  </span>
   682→                  <div className="flex items-center gap-1.5">
   683→                    <MethodBadge method={t.paymentMethod} />
   684→                    {t.status === 'VOIDED' && <StatusBadge status={t.status} />}
   685→                  </div>
   686→                </div>
   687→                <div className="mt-1.5 flex items-center gap-2 text-xs">
   688→                  <span className="min-w-0 flex-1 truncate text-muted-foreground">
   689→                    {t.items.map((it) => it.serviceName).join(', ')}
   690→                  </span>
   691→                  <span className="shrink-0 text-muted-foreground">{t.staffName}</span>
   692→                </div>
   693→              </motion.button>
   694→            ))}
   695→          </div>
   696→
   697→          {/* Load more */}
   698→          <div className="mt-4 space-y-2">
   699→            {listQ.hasNextPage && (
   700→              <Button
   701→                variant="outline"
   702→                className="w-full"
   703→                onClick={() => void listQ.fetchNextPage()}
   704→                disabled={listQ.isFetchingNextPage}
   705→              >
   706→                {listQ.isFetchingNextPage ? (
   707→                  <Loader2 className="animate-spin" aria-hidden />
   708→                ) : null}
   709→                Load more
   710→              </Button>
   711→            )}
   712→            <p className="text-center text-xs text-muted-foreground">
   713→              Showing {rows.length} of {total.toLocaleString('en-IN')} bills
   714→            </p>
   715→          </div>
   716→        </>
   717→      )}
   718→
   719→      <BillDetailDialog
   720→        bill={selected}
   721→        settings={settingsQ.data}
   722→        isAdmin={user?.role === 'ADMIN'}
   723→        onClose={() => setSelected(null)}
   724→      />
   725→    </div>
   726→  )
   727→}
